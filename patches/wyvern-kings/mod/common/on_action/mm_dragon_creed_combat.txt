# root is the deceased; the engine only supplies killer when it is known.
on_death = { on_actions = { mm_creed_combat_death_reward } }
mm_creed_combat_death_reward = {
    effect = {
        if = {
            limit = {
                death_reason = death_battle
                exists = scope:killer
                scope:killer = { is_alive = yes rite = rite:mm_dragon_sky }
            }
            scope:killer = { change_spiritual_fulfillment = 10 }
        }
        if = {
            limit = {
                OR = { death_reason = death_execution death_reason = death_execution_killer death_reason = death_execution_blood_eagle death_reason = death_execution_blood_eagle_killer death_reason = death_execution_blot }
                OR = { rite = rite:mm_dragon_fire rite = rite:mm_dragon_earth rite = rite:mm_dragon_sea rite = rite:mm_dragon_sky rite = rite:mm_dragon_white }
                exists = scope:killer
                scope:killer = { is_alive = yes rite = rite:mm_dragon_black }
            }
            scope:killer = { change_spiritual_fulfillment = 10 }
        }
    }
}
