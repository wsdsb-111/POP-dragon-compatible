mm_royal_queue_daily_effect = {
    if = {
        limit = { is_alive = yes mm_royal_is_king = yes NOT = { has_character_flag = mm_royal_daily_queued } }
        add_character_flag = { flag = mm_royal_daily_queued days = 2 }
        trigger_event = { id = mm_royal.0001 days = 1 }
    }
}
mm_royal_awaken_effect = {
    if = {
        limit = { is_alive = yes mm_royal_is_king = yes has_trait = mm_royal_torpor }
        remove_trait = mm_royal_torpor
        add_character_flag = mm_royal_awakened
        myv_refresh_wyvern_prowess_effect = yes
        mm_royal_queue_daily_effect = yes
    }
}

mm_royal_sync_throne_identity_trait_effect = {
    remove_trait = mm_royal_fire_king
    remove_trait = mm_royal_earth_king
    remove_trait = mm_royal_sea_king
    remove_trait = mm_royal_sky_king
    if = { limit = { has_variable = mm_royal_throne var:mm_royal_throne = 1 } add_trait = mm_royal_fire_king }
    if = { limit = { has_variable = mm_royal_throne var:mm_royal_throne = 2 } add_trait = mm_royal_earth_king }
    if = { limit = { has_variable = mm_royal_throne var:mm_royal_throne = 3 } add_trait = mm_royal_sea_king }
    if = { limit = { has_variable = mm_royal_throne var:mm_royal_throne = 4 } add_trait = mm_royal_sky_king }
}
