    if = { limit = { has_trait_xp = { trait = myv_wyvern_wing_injuries track = myv_wyvern_wound_left_wing value > 0 } } add_trait_xp = { trait = myv_wyvern_wing_injuries track = myv_wyvern_wound_left_wing value = -1 } }
    if = { limit = { has_trait_xp = { trait = myv_wyvern_wing_injuries track = myv_wyvern_wound_right_wing value > 0 } } add_trait_xp = { trait = myv_wyvern_wing_injuries track = myv_wyvern_wound_right_wing value = -1 } }
    if = { limit = { has_trait_xp = { trait = myv_wyvern_hindleg_injuries track = myv_wyvern_wound_left_hindleg value > 0 } } add_trait_xp = { trait = myv_wyvern_hindleg_injuries track = myv_wyvern_wound_left_hindleg value = -1 } }
    if = { limit = { has_trait_xp = { trait = myv_wyvern_hindleg_injuries track = myv_wyvern_wound_right_hindleg value > 0 } } add_trait_xp = { trait = myv_wyvern_hindleg_injuries track = myv_wyvern_wound_right_hindleg value = -1 } }
    if = { limit = { has_trait_xp = { trait = myv_wyvern_tail_injury value > 0 } } add_trait_xp = { trait = myv_wyvern_tail_injury value = -1 } }

    if = {
        limit = {
            has_trait = myv_wyvern_eye_injuries
            NOT = { has_trait_xp = { trait = myv_wyvern_eye_injuries track = myv_wyvern_wound_left_eye value > 0 } }
            NOT = { has_trait_xp = { trait = myv_wyvern_eye_injuries track = myv_wyvern_wound_right_eye value > 0 } }
        }
        remove_trait = myv_wyvern_eye_injuries
    }
    if = {
        limit = {
            has_trait = myv_wyvern_wing_injuries
            NOT = { has_trait_xp = { trait = myv_wyvern_wing_injuries track = myv_wyvern_wound_left_wing value > 0 } }
            NOT = { has_trait_xp = { trait = myv_wyvern_wing_injuries track = myv_wyvern_wound_right_wing value > 0 } }
        }
        remove_trait = myv_wyvern_wing_injuries
    }
    if = {
        limit = {
            has_trait = myv_wyvern_hindleg_injuries
            NOT = { has_trait_xp = { trait = myv_wyvern_hindleg_injuries track = myv_wyvern_wound_left_hindleg value > 0 } }
            NOT = { has_trait_xp = { trait = myv_wyvern_hindleg_injuries track = myv_wyvern_wound_right_hindleg value > 0 } }
        }
        remove_trait = myv_wyvern_hindleg_injuries
    }
    if = {
        limit = {
            has_trait = myv_wyvern_tail_injury
            NOT = { has_trait_xp = { trait = myv_wyvern_tail_injury value > 0 } }
        }
        remove_trait = myv_wyvern_tail_injury
    }
    }
    # Recovery can cross a grade boundary, so keep the native value used by
    # knights and hunt advantage calculations synchronized immediately.
    myv_refresh_wyvern_prowess_effect = yes
}
