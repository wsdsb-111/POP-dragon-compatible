ruler_exit_torpor = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_torpor.dds"
	}

	desc = ruler_exit_torpor_desc
	decision_group_type = pod_splat

	ai_check_interval = 120

	is_shown = {
		NOT = { mm_royal_is_king = yes }
		is_playable_character = yes
		OR = {
		has_trait = torpor
		AND = {
			is_ai = yes
			has_character_modifier = trapped_in_arcadia_modifier
		}
		AND = {
			is_ai = yes
			has_trait = wight
		}
		}
	}

	is_valid = {
		NOT = { mm_royal_is_king = yes }	
		OR = {
		has_trait = torpor
		AND = {
			is_ai = yes
			has_character_modifier = trapped_in_arcadia_modifier
		}
		AND = {
			is_ai = yes
			has_trait = wight
		}		
		}
	}

	is_valid_showing_failures_only = {
	}

	effect = {
		if = {
			limit = {
				NOT = { mm_royal_is_king = yes }
			}
		if = {
			limit = {
				has_trait = torpor
			}
		if = {
			limit = { POD_is_not_trapped_in_umbra_trigger = no }
			POD_depose_effect = yes
		}
		else = {
			remove_trait = torpor
		}
		}
		
		if = {
			limit = {
				is_ai = yes
				has_character_modifier = trapped_in_arcadia_modifier
			}
		if = {
			limit = {
				is_landed = yes 
			}
			random_neighboring_and_across_water_top_liege_realm_owner = {
					limit = {
						NOT = { has_trait = fae }
						NOT = { has_trait = changeling }
					}
				save_temporary_scope_as = canon_inheritor
				}
			POD_transfer_all_titles_to_character = {
				CHARACTER = scope:canon_inheritor
			}
		}
		else = { POD_depose_effect = yes }
		}

		if = {
			limit = {
				is_ai = yes
				has_trait = wight
			}
		if = {
			limit = { POD_is_not_trapped_in_umbra_trigger = no }
			POD_depose_effect = yes
		}
		else = {
			remove_trait = wight
		}
		}
		}
	}	
	
	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}
