mm_royal_power_factor = {
    value = mm_royal_power_cap
    divide = mm_pod_elder_reference_prowess
}

myv_wyvern_target_prowess = {
    value = 0
    if = {
        limit = { NOT = { has_trait = mm_royal_torpor } }
        value = myv_wyvern_size_prowess
        multiply = mm_wyvern_pod_size_power_factor
        multiply = myv_wyvern_physique_prowess_multiplier
        multiply = myv_wyvern_temperament_prowess_multiplier
        if = {
            limit = { mm_royal_is_king = yes }
            multiply = mm_royal_power_factor
        }
        if = { limit = { has_character_modifier = myv_wyvern_enraged } multiply = 1.1 }

        # Cap the fully fed size-based value before condition penalties.
        max = mm_royal_power_cap
        multiply = myv_wyvern_wing_injury_prowess_multiplier
        multiply = myv_wyvern_hindleg_injury_prowess_multiplier
        multiply = myv_wyvern_tail_injury_prowess_multiplier
        multiply = mm_wyvern_hunger_power_multiplier
        multiply = myv_wyvern_malnutrition_prowess_multiplier
        if = {
            limit = { has_variable = myv_wyvern_hunger var:myv_wyvern_hunger <= 0 }
            value = 0
        }
        round = yes
        min = 0
        max = mm_royal_power_cap
    }
}
