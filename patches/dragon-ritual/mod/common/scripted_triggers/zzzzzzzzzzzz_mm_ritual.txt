mm_ritual_sacrifice_valid = {
    myv_is_living_wyvern = yes
    mm_dragon_is_true_dragon_parent_trigger = yes
    mm_royal_is_king = no
    NOT = { has_trait = mm_dragon_half }
    NOT = { has_character_flag = mm_wyvern_human_form }
    OR = {
        AND = { has_trait = myv_wyvern_tamed has_relation_myv_wyvern_bonded_character = $OWNER$ }
        AND = { mm_wyvern_capture_valid_for_captor_trigger = { CAPTOR = $OWNER$ } has_character_flag = mm_wyvern_capture_stable_cage }
    }
}
mm_ritual_selected_valid = {
    exists = var:mm_ritual_selected_sacrifice
    var:mm_ritual_selected_sacrifice = {
        mm_ritual_sacrifice_valid = { OWNER = root }
        OR = {
            AND = { has_variable = mm_royal_creation_throne var:mm_royal_creation_throne = $THRONE$ }
            AND = {
                NOT = { has_variable = mm_royal_creation_throne }
                OR = {
                    AND = { exists = mother mother = { mm_dragon_is_true_dragon_parent_trigger = yes var:mm_royal_creation_throne = $THRONE$ } }
                    AND = { exists = real_father real_father = { mm_dragon_is_true_dragon_parent_trigger = yes var:mm_royal_creation_throne = $THRONE$ } }
                }
                NOT = { exists = mother mother = { has_variable = mm_royal_creation_throne NOT = { var:mm_royal_creation_throne = $THRONE$ } } }
                NOT = { exists = real_father real_father = { has_variable = mm_royal_creation_throne NOT = { var:mm_royal_creation_throne = $THRONE$ } } }
            }
        }
    }
}
