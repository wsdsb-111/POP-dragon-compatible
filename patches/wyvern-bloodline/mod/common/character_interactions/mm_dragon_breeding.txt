# Two explicit character clicks use the native interaction UI; no unverified
# secondary-recipient picker or hidden random mate selection is involved.
mm_dragon_select_stable_mother_interaction = {
	category = interaction_category_friendly
	filter_tags = { myv_wyvern_interaction }
	icon = icon_marriage
	common_interaction = yes
	auto_accept = yes
	desc = mm_dragon_select_stable_mother_interaction_desc
	is_shown = {
		scope:actor = { is_ai = no is_alive = yes is_imprisoned = no is_incapable = no }
		scope:recipient = { mm_dragon_stable_mother_valid_trigger = { KEEPER = scope:actor } }
	}
	on_accept = {
		if = {
			limit = { scope:recipient = { mm_dragon_stable_mother_valid_trigger = { KEEPER = scope:actor } } }
			scope:actor = { set_variable = { name = mm_dragon_selected_mother value = scope:recipient } }
		}
	}
}
mm_dragon_force_stable_breeding_interaction = {
	category = interaction_category_friendly
	filter_tags = { myv_wyvern_interaction }
	icon = icon_marriage
	common_interaction = yes
	auto_accept = yes
	desc = mm_dragon_force_stable_breeding_interaction_desc
	is_shown = {
		scope:actor = { is_ai = no is_alive = yes is_imprisoned = no is_incapable = no has_variable = mm_dragon_selected_mother }
		scope:recipient = { mm_dragon_stable_pair_valid_trigger = { MOTHER = scope:actor.var:mm_dragon_selected_mother KEEPER = scope:actor } }
	}
	is_valid_showing_failures_only = {
		scope:recipient = { mm_dragon_stable_pair_valid_trigger = { MOTHER = scope:actor.var:mm_dragon_selected_mother KEEPER = scope:actor } }
	}
	on_accept = {
		if = {
			limit = {
				scope:actor = { is_alive = yes is_imprisoned = no is_incapable = no has_variable = mm_dragon_selected_mother }
				scope:recipient = { mm_dragon_stable_pair_valid_trigger = { MOTHER = scope:actor.var:mm_dragon_selected_mother KEEPER = scope:actor } }
			}
			scope:actor.var:mm_dragon_selected_mother = {
				save_scope_as = myv_wyvern_egg_mother
				save_scope_as = myv_wyvern_egg_creator
				location = { save_scope_as = myv_wyvern_egg_location }
			}
			scope:recipient = { save_scope_as = myv_wyvern_egg_father }
			clear_saved_scope = myv_wyvern_egg
			scope:actor = { myv_create_stable_bred_wyvern_egg_artifact_effect = yes }
			if = {
				limit = { exists = scope:myv_wyvern_egg }
				scope:myv_wyvern_egg_mother = { add_character_flag = { flag = mm_dragon_forced_breeding_cooldown years = 1 } }
				scope:actor = { remove_variable = mm_dragon_selected_mother }
			}
		}
	}
}
