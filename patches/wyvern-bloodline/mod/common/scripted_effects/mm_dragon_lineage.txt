# TARGET character scope; acquired branches are visible but never inheritance input.
mm_dragon_refresh_branch_traits_effect = {
    # Recompute only when every exact source carries usable branch evidence.
    # Unknown legacy records and acquired powers are preserved.
    if = {
        limit = {
            mm_dragon_is_royal_parent_trigger = no
            has_variable_list = mm_dragon_origins
            NOT = { any_in_list = { variable = mm_dragon_origins OR = {
                mm_dragon_source_has_usable_branch_trigger = no
                AND = { this = prev mm_dragon_source_has_defined_branch_trigger = no }
            } } }
        }
        save_scope_as = mm_dragon_branch_projection_target
        remove_variable = mm_dragon_bio_fire
        remove_variable = mm_dragon_bio_sea
        remove_variable = mm_dragon_bio_sky
        remove_variable = mm_dragon_bio_earth
        remove_variable = mm_dragon_bio_white
        remove_variable = mm_dragon_bio_black
        every_in_list = {
            variable = mm_dragon_origins
            mm_dragon_project_biological_branch_effect = { TARGET = scope:mm_dragon_branch_projection_target }
        }
    }
    if = {
        limit = { mm_dragon_is_royal_parent_trigger = yes }
        remove_trait = mm_dragon_branch_fire
        remove_trait = mm_dragon_branch_sea
        remove_trait = mm_dragon_branch_sky
        remove_trait = mm_dragon_branch_earth
        remove_trait = mm_dragon_branch_white
        remove_trait = mm_dragon_branch_black
    }
    else = {
        if = { limit = { OR = { has_variable = mm_dragon_bio_fire has_variable = mm_dragon_absorbed_fire AND = { has_variable = mm_royal_creation_throne var:mm_royal_creation_throne = 1 } } } add_trait = mm_dragon_branch_fire }
        else = { remove_trait = mm_dragon_branch_fire }
        if = { limit = { OR = { has_variable = mm_dragon_bio_sea has_variable = mm_dragon_absorbed_sea AND = { has_variable = mm_royal_creation_throne var:mm_royal_creation_throne = 3 } } } add_trait = mm_dragon_branch_sea }
        else = { remove_trait = mm_dragon_branch_sea }
        if = { limit = { OR = { has_variable = mm_dragon_bio_sky has_variable = mm_dragon_absorbed_sky AND = { has_variable = mm_royal_creation_throne var:mm_royal_creation_throne = 4 } } } add_trait = mm_dragon_branch_sky }
        else = { remove_trait = mm_dragon_branch_sky }
        if = { limit = { OR = { has_variable = mm_dragon_bio_earth has_variable = mm_dragon_absorbed_earth AND = { has_variable = mm_royal_creation_throne var:mm_royal_creation_throne = 2 } } } add_trait = mm_dragon_branch_earth }
        else = { remove_trait = mm_dragon_branch_earth }
        if = { limit = { OR = { has_variable = mm_dragon_bio_white has_variable = mm_dragon_absorbed_white } } add_trait = mm_dragon_branch_white }
        else = { remove_trait = mm_dragon_branch_white }
        if = { limit = { OR = { has_variable = mm_dragon_bio_black has_variable = mm_dragon_absorbed_black } } add_trait = mm_dragon_branch_black }
        else = { remove_trait = mm_dragon_branch_black }
    }
}

mm_dragon_consider_parent_effect = {

        if = { limit = { exists = scope:real_father scope:real_father = { mm_dragon_has_inheritable_blood_trigger = yes has_trait = mm_dragon_black_blood } } scope:real_father = { mm_dragon_copy_biological_origins_effect = { CHILD = scope:mm_dragon_lineage_child } } }
        # A single black-blood parent supplies ancestry, not an inherited tier.
        # If the other parent has no mixed tier, the ordinary baseline is C.
        if = {
            limit = { NOT = { has_variable = mm_dragon_lineage_generation } OR = {
                AND = { exists = scope:mother scope:mother = { mm_dragon_has_inheritable_blood_trigger = yes has_trait = mm_dragon_black_blood } }
                AND = { exists = scope:real_father scope:real_father = { mm_dragon_has_inheritable_blood_trigger = yes has_trait = mm_dragon_black_blood } }
            } }
            set_variable = { name = mm_dragon_lineage_generation value = 4 }
        }
        # Baseline is one grade below the stronger parent; bonus is one above it, capped at S.
        if = { limit = { has_variable = mm_dragon_lineage_generation } set_variable = { name = mm_dragon_stronger_parent_generation value = var:mm_dragon_lineage_generation } change_variable = { name = mm_dragon_stronger_parent_generation subtract = 2 } if = { limit = { var:mm_dragon_stronger_parent_generation < 1 } set_variable = { name = mm_dragon_stronger_parent_generation value = 1 } } }

        # True dragon parentage fixes S/first generation, with no human reroll.
        if = {
            limit = {
                NOT = { OR = {
                    AND = { exists = scope:mother scope:mother = { mm_dragon_is_true_dragon_parent_trigger = yes } }
                    AND = { exists = scope:real_father scope:real_father = { mm_dragon_is_true_dragon_parent_trigger = yes } }
