myv_wyvern_character_window = {
    scope = character

    is_shown = {
        myv_is_wyvern = yes
        NOT = { has_trait = mm_asc_dragon_body }
        NOT = { has_character_flag = mm_wyvern_human_form }
    }
}

# The native birth-name controller always samples the culture of the character
# being renamed. These actions select that source culture before it rolls.
# The hatching event restores the original culture on confirmation.
myv_hatchling_use_wyvern_culture_for_name = {
	scope = character

	is_valid = {
		exists = scope:myv_hatchling
		scope:myv_hatchling = { myv_is_living_wyvern = yes }
	}

	effect = {
		scope:myv_hatchling = {
			if = { limit = { NOT = { has_variable = mm_dragon_name_original_culture } } set_variable = { name = mm_dragon_name_original_culture value = culture } }
			set_culture = culture:myv_wyvern_culture
		}
	}
}

myv_hatchling_use_player_culture_for_name = {
	scope = character

	is_valid = {
		exists = scope:myv_hatchling
		scope:myv_hatchling = { myv_is_living_wyvern = yes }
	}

	effect = {
		scope:myv_hatchling = {
			if = { limit = { NOT = { has_variable = mm_dragon_name_original_culture } } set_variable = { name = mm_dragon_name_original_culture value = culture } }
			set_culture = root.culture
		}
	}
}

# A resting wyvern is shown inside its lair. Roaming wyverns use the terrain
# at their current location; all unlisted terrain falls back to the grassland
# scene (including forests and coasts).
myv_wyvern_scene_cave = {
    scope = character
    is_shown = {
        NOT = { has_character_flag = myv_wyvern_roaming }
        NOT = { has_trait = myv_wyvern_tamed }
    }
}

myv_wyvern_scene_stable = {
	scope = character
	is_shown = {
		has_character_flag = myv_wyvern_in_stable
	}
}

myv_wyvern_scene_snow = {
    scope = character
    is_shown = {
		NOT = { has_character_flag = myv_wyvern_in_stable }
        OR = {
            has_character_flag = myv_wyvern_roaming
            has_trait = myv_wyvern_tamed
        }
        exists = location
        location = { terrain = taiga }
    }
}

myv_wyvern_scene_mountains = {
    scope = character
    is_shown = {
		NOT = { has_character_flag = myv_wyvern_in_stable }
        OR = {
            has_character_flag = myv_wyvern_roaming
            has_trait = myv_wyvern_tamed
        }
        exists = location
        location = {
            OR = {
                terrain = hills
                terrain = mountains
            }
        }
    }
}

myv_wyvern_scene_desert = {
    scope = character
    is_shown = {
		NOT = { has_character_flag = myv_wyvern_in_stable }
        OR = {
            has_character_flag = myv_wyvern_roaming
            has_trait = myv_wyvern_tamed
        }
        exists = location
        location = {
            OR = {
                terrain = desert
                terrain = desert_mountains
                terrain = drylands
                terrain = oasis
            }
        }
    }
}

myv_wyvern_scene_grassland = {
    scope = character
    is_shown = {
		NOT = { has_character_flag = myv_wyvern_in_stable }
        OR = {
            has_character_flag = myv_wyvern_roaming
            has_trait = myv_wyvern_tamed
        }
        OR = {
            NOT = { exists = location }
            location = {
                NOT = {
                    OR = {
                        terrain = taiga
                        terrain = hills
                        terrain = mountains
                        terrain = desert
                        terrain = desert_mountains
                        terrain = drylands
                        terrain = oasis
                    }
                }
            }
        }
    }
}
