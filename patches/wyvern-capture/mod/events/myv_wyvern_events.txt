# A wyvern has reached this county and torn through the local herds. The loss
# is applied by the dispatching effect so it resolves immediately even for AI
# rulers and cannot be postponed by leaving a player event open.
myv_wyvern.0100 = {
    type = character_event
    trigger = {
        exists = scope:myv_raid_victim
        this = scope:myv_raid_victim
        OR = {
            exists = scope:myv_raid_debug_override
            myv_is_actual_wyvern_raid_recipient = { COUNTY = scope:myv_raided_county }
        }
    }
    override_background = { reference = myv_wyvern_hungry_raid }
    title = myv_wyvern.0100.t
    desc = {
        first_valid = {
            triggered_desc = {
                trigger = { exists = scope:myv_raid_court_victim }
                desc = myv_wyvern.0100.desc_court_death
            }
            triggered_desc = {
                trigger = { exists = scope:myv_raid_debt_branch }
                desc = myv_wyvern.0100.desc_empty_court
            }
            desc = myv_wyvern.0100.desc
        }
    }
    theme = martial

    left_portrait = {
        character = scope:myv_raid_court_victim
        trigger = { exists = scope:myv_raid_court_victim }
        animation = fear
        animate_if_dead = yes
    }

    option = {
        name = myv_wyvern.0100.a
        show_as_tooltip = {
            if = {
                limit = { NOT = { exists = scope:myv_raid_debt_branch } }
                remove_short_term_gold = 50
            }
            if = {
                limit = { exists = scope:myv_raid_court_victim }
                custom_tooltip = myv_wyvern.0100.court_death_tt
            }
            add_prestige = -100
        }
    }
}

# The annual egg pulse creates the hatchling before opening this notification.
# scope:myv_new_wyvern is preserved from the creation effect.
myv_wyvern.0300 = {
	type = character_event
	window = myv_wyvern_feeding_event
	title = myv_wyvern.0300.t
	desc = myv_wyvern.0300.desc
	theme = family
	override_background = { reference = myv_wyvern_hatching }

	immediate = {
		# The feeding and hatching scenes share one custom window. This flag is
		# event-specific so an ordinary feeding scene with a young wyvern keeps
		# the normal feeding camera, placement and hover glow.
		scope:myv_new_wyvern = {
			add_character_flag = myv_wyvern_hatching_event_portrait
		}
	}

	right_portrait = {
		character = scope:myv_new_wyvern
		animation = myv_wyvern_main
	}

	widgets = {
		widget = {
			is_shown = { always = yes }
			gui = "myv_wyvern_hatchling_name"
			container = "dynamic_birth_name"
			controller = name_character
			setup_scope = {
				scope:myv_new_wyvern = { save_scope_as = name_character_target }
			}
		}
	}

	option = {
		name = myv_wyvern.0300.a
		# A player-culture randomization temporarily changes only the culture
		# consulted by the name controller; confirmation restores its original culture.
		scope:myv_new_wyvern = {
			remove_character_flag = myv_wyvern_hatching_event_portrait
			if = {
                limit = { exists = var:mm_dragon_name_original_culture }
                set_culture = var:mm_dragon_name_original_culture
                remove_variable = mm_dragon_name_original_culture
            }
		}
	}
}

# Resolve the ambush after the option is selected. Option tooltip previews do
# not retain a variable set by an earlier random effect in the same option.
myv_wyvern.0207 = {
	type = character_event
	hidden = yes
	immediate = {
		set_variable = { name = myv_slay_wyvern_result value = 0 }
		random = {
			chance = myv_slay_ambush_chance
			set_variable = { name = myv_slay_wyvern_result value = 1 }
		}
		scope:myv_slay_wyvern_target = {
			if = {
				limit = { has_character_modifier = myv_wyvern_enraged }
				remove_character_modifier = myv_wyvern_enraged
			}
			add_character_modifier = {
				modifier = myv_wyvern_enraged
				months = 3
			}
			myv_refresh_wyvern_prowess_effect = yes
		}
		if = {
			limit = { var:myv_slay_wyvern_result = 1 }
			set_variable = { name = myv_slay_target_part value = 1 }
			set_variable = { name = myv_slay_wound_amount value = 75 }
			myv_slay_apply_targeted_wound_effect = yes
			set_variable = { name = myv_slay_previous_outcome value = 5 }
		}
		else = { set_variable = { name = myv_slay_previous_outcome value = 6 } }
		trigger_event = myv_wyvern.0209
	}
}

# First-exchange router. It is shared by the face-to-face opening and both
# outcomes of the sleeping-wyvern ambush, and deliberately uses the lair pool.
myv_wyvern.0209 = {
	type = character_event
	hidden = yes
	immediate = {
		if = {
			limit = { NOT = { scope:myv_slay_wyvern_target = { myv_is_living_wyvern = yes } } }
			trigger_event = myv_wyvern.0204
		}
		else = {
			myv_slay_choose_lair_action_effect = yes
			if = { limit = { var:myv_slay_wyvern_action = 1 } trigger_event = myv_wyvern.0211 }
			else_if = { limit = { var:myv_slay_wyvern_action = 2 } trigger_event = myv_wyvern.0212 }
			else_if = { limit = { var:myv_slay_wyvern_action = 4 } trigger_event = myv_wyvern.0214 }
			else_if = { limit = { var:myv_slay_wyvern_action = 5 } trigger_event = myv_wyvern.0215 }
			else = { trigger_event = myv_wyvern.0204 }
		}
	}
}

# Central result router for every wyvern hit, including a failed withdrawal.
# 4 = light wound, 2 = severe/escalated wound, 3 = death.
myv_wyvern.0208 = {
	type = character_event
	hidden = yes
	immediate = {
		if = {
			limit = { var:myv_slay_wyvern_result = 4 }
			add_trait = wounded_1
			add_prestige = -50
			if = {
				limit = { var:myv_slay_retreat_failed = 1 }
				set_variable = { name = myv_slay_previous_outcome value = 8 }
				remove_variable = myv_slay_retreat_failed
			}
			else = { set_variable = { name = myv_slay_previous_outcome value = 7 } }
			trigger_event = myv_wyvern.0229
		}
		else_if = { limit = { var:myv_slay_wyvern_result = 2 } trigger_event = myv_wyvern.0202 }
		else_if = { limit = { var:myv_slay_wyvern_result = 3 } trigger_event = myv_wyvern.0203 }
		else = { trigger_event = myv_wyvern.0204 }
	}
}

	type = character_event
	hidden = yes
	immediate = {
		set_variable = { name = myv_slay_wyvern_result value = 0 }

		if = {
			limit = { NOT = { scope:myv_slay_wyvern_target = { myv_is_living_wyvern = yes } } }
			trigger_event = myv_wyvern.0204
		}
		else_if = {
			limit = {
				scope:myv_slay_wyvern_target = {
					OR = {
						has_trait_xp = { trait = myv_wyvern_eye_injuries track = myv_wyvern_wound_left_eye value >= 100 }
						has_trait_xp = { trait = myv_wyvern_eye_injuries track = myv_wyvern_wound_right_eye value >= 100 }
						has_trait_xp = { trait = myv_wyvern_wing_injuries track = myv_wyvern_wound_left_wing value >= 100 }
						has_trait_xp = { trait = myv_wyvern_wing_injuries track = myv_wyvern_wound_right_wing value >= 100 }

						has_trait_xp = { trait = myv_wyvern_hindleg_injuries track = myv_wyvern_wound_right_hindleg value >= 100 }
						has_trait_xp = { trait = myv_wyvern_tail_injury value >= 100 }
					}
				}
			}
			trigger_event = myv_wyvern.0250
		}
		else_if = {
			limit = { var:myv_slay_round >= 5 }
			if = { limit = { var:myv_slay_hits >= 3 } trigger_event = myv_wyvern.0250 }
			else = { trigger_event = myv_wyvern.0252 }
		}
		else = {
			change_variable = { name = myv_slay_round add = 1 }
			if = {
				limit = { var:myv_slay_outdoor_battle = 1 }
				myv_slay_choose_action_effect = yes
			}
			else = { myv_slay_choose_lair_action_effect = yes }
			if = { limit = { var:myv_slay_wyvern_action = 1 } trigger_event = myv_wyvern.0211 }
			else_if = { limit = { var:myv_slay_wyvern_action = 2 } trigger_event = myv_wyvern.0212 }
			else_if = { limit = { var:myv_slay_wyvern_action = 3 } trigger_event = myv_wyvern.0213 }
			else_if = { limit = { var:myv_slay_wyvern_action = 4 } trigger_event = myv_wyvern.0214 }
			else_if = { limit = { var:myv_slay_wyvern_action = 5 } trigger_event = myv_wyvern.0215 }

		}
	}
