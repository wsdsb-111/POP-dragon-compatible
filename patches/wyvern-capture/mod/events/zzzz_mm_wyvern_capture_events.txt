mm_wyvern_capture.0002 = {
	type = character_event
	window = myv_wyvern_feeding_event
	override_background = { reference = myv_wyvern_stable }
	title = mm_wyvern_capture.0002.t
	desc = mm_wyvern_capture.0002.desc
	theme = diplomacy
	left_portrait = { character = root animation = happiness }
	right_portrait = { character = scope:mm_wyvern_capture_scheme_target animation = myv_wyvern_main }
	option = {
		name = mm_wyvern_capture.0002.a
		custom_tooltip = mm_wyvern_capture.tame_success_tt
		hidden_effect = {
		if = {
			limit = {
				is_alive = yes
				scope:mm_wyvern_capture_scheme_target = {
					mm_wyvern_capture_valid_for_captor_trigger = { CAPTOR = root }
					var:mm_wyvern_taming_progress >= 100
				}
			}
			scope:mm_wyvern_capture_scheme_target = { mm_wyvern_capture_tame_effect = { KEEPER = root } }
			myv_notify_wyvern_success_effect = {
				ACTOR = root
				WYVERN = scope:mm_wyvern_capture_scheme_target
				TITLE = myv_wyvern_bond_success_news_title
				DESC = myv_wyvern_bond_success_news_desc
			}
		}
		scope:mm_wyvern_capture_scheme = { end_scheme = yes }
		}
	}
	after = {
		clear_saved_scope = mm_wyvern_capture_scheme
		clear_saved_scope = mm_wyvern_capture_scheme_target
		clear_saved_scope = mm_wyvern_capture_scheme_owner
		clear_saved_scope = mm_wyvern_capture_scheme_successful
	}
}

mm_wyvern_capture.0003 = {
	type = character_event
	window = myv_wyvern_feeding_event
	override_background = { reference = myv_wyvern_stable }
	title = mm_wyvern_capture.0003.t
	desc = mm_wyvern_capture.0003.desc
	theme = martial
	left_portrait = { character = root animation = fear }
	right_portrait = { character = scope:mm_wyvern_capture_scheme_target animation = myv_wyvern_main }
	option = {
		name = mm_wyvern_capture.0003.a
		if = {
			limit = {
				is_alive = yes
				scope:mm_wyvern_capture_scheme_target = {
					mm_wyvern_capture_valid_for_captor_trigger = { CAPTOR = root }
					var:mm_wyvern_taming_progress >= 100
				}
			}
			random_list = {
				5 = { death = { death_reason = death_myv_failed_wyvern_bonding killer = scope:mm_wyvern_capture_scheme_target } }
				25 = {
					if = {
						limit = { NOT = { has_trait = wounded_3 } }
						if = {
							limit = { OR = { has_trait = wounded_1 has_trait = wounded_2 } }
							if = { limit = { has_trait = wounded_1 } remove_trait = wounded_1 }
							if = { limit = { has_trait = wounded_2 } remove_trait = wounded_2 }
							add_trait = wounded_3
						}
						else = { add_trait = wounded_2 }
					}
				}
				70 = {
					if = {
						limit = { NOT = { has_trait = wounded_3 } }
						if = { limit = { has_trait = wounded_2 } remove_trait = wounded_2 add_trait = wounded_3 }
						else_if = { limit = { has_trait = wounded_1 } remove_trait = wounded_1 add_trait = wounded_2 }
						else = { add_trait = wounded_1 }
					}
				}
			}
		}
		scope:mm_wyvern_capture_scheme = { end_scheme = yes }
	}
	after = {
		clear_saved_scope = mm_wyvern_capture_scheme
		clear_saved_scope = mm_wyvern_capture_scheme_target
		clear_saved_scope = mm_wyvern_capture_scheme_owner
		clear_saved_scope = mm_wyvern_capture_scheme_successful
	}
}
