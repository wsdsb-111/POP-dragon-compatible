mm_wyvern_apply_dragon_weapon_affixes_effect = {
	if = {
		limit = {
			has_variable = mm_wyvern_weapon_kind
			NOT = { has_variable = mm_wyvern_weapon_affixes_generated }
		}
		if = {
			limit = { var:mm_wyvern_weapon_kind = flag:sword }
			if = { limit = { rarity = masterwork } mm_wyvern_roll_dragon_weapon_affix_effect = yes }
			if = {
				limit = { rarity = famed }
				mm_wyvern_roll_dragon_weapon_affix_effect = yes
				mm_wyvern_roll_dragon_weapon_affix_effect = yes
			}
			if = {
				limit = { rarity = illustrious }
				mm_wyvern_roll_dragon_weapon_affix_effect = yes
				mm_wyvern_roll_dragon_weapon_affix_effect = yes
				mm_wyvern_roll_dragon_weapon_affix_effect = yes
				mm_wyvern_roll_dragon_weapon_affix_effect = yes
			}
			set_variable = { name = mm_wyvern_weapon_affixes_generated value = 1 }
		}
		if = {
			limit = { var:mm_wyvern_weapon_kind = flag:bow }
			if = { limit = { rarity = famed } mm_wyvern_roll_dragon_weapon_affix_effect = yes }
			if = {
				limit = { rarity = illustrious }
				mm_wyvern_roll_dragon_weapon_affix_effect = yes
				mm_wyvern_roll_dragon_weapon_affix_effect = yes
			}
			set_variable = { name = mm_wyvern_weapon_affixes_generated value = 1 }
		}
	}
}
