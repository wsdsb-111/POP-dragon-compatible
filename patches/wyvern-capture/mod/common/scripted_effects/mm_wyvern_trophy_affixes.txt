# Each artifact rolls independently, without replacement between categories.
mm_wyvern_apply_trophy_affixes_effect = {
	if = {
		limit = { NOT = { has_variable = mm_wyvern_trophy_affixes_generated } }
		remove_artifact_modifier = artifact_placeholder_modifier
		remove_artifact_modifier = artifact_monthly_minor_prestige_1_modifier
		mm_wyvern_roll_trophy_affix_effect = yes
		mm_wyvern_roll_trophy_affix_effect = yes
		mm_wyvern_roll_trophy_affix_effect = yes
		if = { limit = { rarity = illustrious } mm_wyvern_roll_trophy_affix_effect = yes }
		set_variable = { name = mm_wyvern_trophy_affixes_generated value = 1 }
	}
}
