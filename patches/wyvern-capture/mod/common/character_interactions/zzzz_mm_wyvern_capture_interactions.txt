# Captured wild wyverns cannot be selected through the original direct hunt or
# debug raid entry points. The remaining interaction flow is unchanged.
myv_attempt_slay_wyvern_interaction = {
	category = interaction_category_hostile
	filter_tags = { myv_wyvern_interaction }
	icon = icon_combat
	interface_priority = 100
	common_interaction = yes
	auto_accept = yes
	use_diplomatic_range = yes
	cooldown_against_recipient = { years = 1 }
	desc = myv_attempt_slay_wyvern_interaction_desc

	is_shown = {
		scope:actor = { is_ai = no is_alive = yes myv_is_wyvern = no }
		scope:recipient = {
			myv_is_living_wyvern = yes
			NOT = { has_trait = myv_wyvern_tamed }
			NOT = { has_variable = mm_wyvern_captor }
		}
		scope:actor != scope:recipient
	}

	is_valid_showing_failures_only = {
		scope:actor = { is_imprisoned = no is_incapable = no }
		scope:recipient = {
			myv_is_living_wyvern = yes
			NOT = { has_trait = myv_wyvern_tamed }
			NOT = { has_variable = mm_wyvern_captor }
		}
	}

	on_accept = {
		scope:actor = {
			if = {
				limit = { scope:recipient = { myv_is_living_wild_wyvern = yes } }
				scope:recipient = { save_scope_as = myv_slay_wyvern_target }
				if = {
					limit = { scope:recipient = { has_character_flag = myv_wyvern_roaming } }
					if = { limit = { myv_is_active_wyvern_war_rider = yes } myv_slay_begin_mounted_outdoor_duel_effect = yes }
					else = {
						myv_slay_reset_state_effect = yes
						set_variable = { name = myv_slay_round value = 1 }
						set_variable = { name = myv_slay_hits value = 0 }
						set_variable = { name = myv_slay_previous_outcome value = 0 }
						set_variable = { name = myv_slay_outdoor_battle value = 1 }
						set_variable = { name = myv_slay_wyvern_action value = 3 }
						set_variable = { name = myv_slay_attack_strength value = 1 }
						trigger_event = myv_wyvern.0213
					}
				}
				else = { hidden_effect = { myv_slay_begin_lair_combat_effect = yes } }
			}
		}
	}
}

myv_debug_trigger_wyvern_attack_interaction = {
	category = interaction_debug_main
	filter_tags = { myv_wyvern_interaction }
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes
	common_interaction = no
	icon = icon_combat
	auto_accept = yes
	desc = myv_debug_trigger_wyvern_attack_interaction_desc

	is_shown = {
		debug_only = yes
		scope:actor = { is_alive = yes myv_is_wyvern = no }
		scope:recipient = {
			myv_is_living_wyvern = yes
			NOT = { has_trait = myv_wyvern_tamed }
			NOT = { has_variable = mm_wyvern_captor }
		}
		scope:actor != scope:recipient
	}

	is_valid_showing_failures_only = {
		scope:recipient = {
			myv_is_living_wyvern = yes
			exists = location.county
			NOT = { has_variable = mm_wyvern_captor }
		}
	}

	on_accept = {
		if = {
			limit = { scope:recipient = { myv_is_living_wild_wyvern = yes exists = location.county } }
			scope:recipient = {
				save_scope_as = myv_raiding_wyvern
				location.county = { save_scope_as = myv_raided_county }
				change_variable = { name = myv_wyvern_hunger add = 20 }
				if = { limit = { var:myv_wyvern_hunger > 100 } set_variable = { name = myv_wyvern_hunger value = 100 } }
				myv_wyvern_refresh_hunger_state_effect = yes
			}
			save_scope_value_as = { name = myv_raid_debug_override value = yes }
			scope:actor = { myv_resolve_wyvern_raid_victim_effect = yes }
			clear_saved_scope = myv_raid_debug_override
		}
	}
}

mm_wyvern_feed_captive_interaction = {
	category = interaction_category_friendly
	filter_tags = { myv_wyvern_interaction }
	icon = artifact
	interface_priority = 100
	common_interaction = yes
	auto_accept = yes
	use_diplomatic_range = yes
	desc = mm_wyvern_feed_captive_interaction_desc

	is_shown = {
		scope:actor = { is_ai = no is_alive = yes myv_is_wyvern = no }
		scope:recipient = {
			mm_wyvern_capture_can_feed_trigger = { CAPTOR = scope:actor }
		}
		scope:actor != scope:recipient
	}

	is_valid_showing_failures_only = {
		scope:actor = { is_imprisoned = no is_incapable = no }
		scope:recipient = { mm_wyvern_capture_can_feed_trigger = { CAPTOR = scope:actor } }
	}

	on_accept = {
		scope:actor = {
			if = {
				limit = { scope:recipient = { mm_wyvern_capture_can_feed_trigger = { CAPTOR = scope:actor } } }
				scope:recipient = { save_scope_as = mm_capture_target }
				trigger_event = mm_wyvern_capture.0001
			}
		}
	}
}

mm_wyvern_attempt_tame_captive_interaction = {
	category = interaction_category_friendly
	filter_tags = { myv_wyvern_interaction }
	icon = icon_scheme_befriend
	interface_priority = 99
	common_interaction = yes
	auto_accept = yes
	use_diplomatic_range = yes
	desc = {
		first_valid = {
			triggered_desc = { trigger = { scope:recipient = { has_trait = myv_wyvern_temperament_alert } } desc = mm_wyvern_attempt_tame_captive_interaction_desc_alert }
			triggered_desc = { trigger = { scope:recipient = { has_trait = myv_wyvern_temperament_docile } } desc = mm_wyvern_attempt_tame_captive_interaction_desc_docile }
			triggered_desc = { trigger = { scope:recipient = { has_trait = myv_wyvern_temperament_fierce } } desc = mm_wyvern_attempt_tame_captive_interaction_desc_fierce }
		}
	}

	is_shown = {
		scope:actor = { is_ai = no is_alive = yes myv_is_wyvern = no }
		scope:recipient = {
			mm_wyvern_capture_valid_for_captor_trigger = { CAPTOR = scope:actor }
			var:mm_wyvern_taming_progress >= 100
			OR = {
				has_trait = myv_wyvern_temperament_alert
				has_trait = myv_wyvern_temperament_docile
				has_trait = myv_wyvern_temperament_fierce
			}
		}
		scope:actor != scope:recipient
	}

	is_valid_showing_failures_only = {
		scope:actor = { is_adult = yes is_imprisoned = no is_incapable = no }
		scope:recipient = {
			mm_wyvern_capture_valid_for_captor_trigger = { CAPTOR = scope:actor }
			var:mm_wyvern_taming_progress >= 100
		}
		trigger_if = {
			limit = { scope:recipient = { has_trait = myv_wyvern_temperament_alert } }
			scope:actor = { can_start_scheme = { type = mm_wyvern_tame_alert_captive target_character = scope:recipient } }
		}
		trigger_else_if = {
			limit = { scope:recipient = { has_trait = myv_wyvern_temperament_docile } }
			scope:actor = { can_start_scheme = { type = mm_wyvern_tame_docile_captive target_character = scope:recipient } }
		}
		trigger_else = {
			scope:recipient = { has_trait = myv_wyvern_temperament_fierce }
			scope:actor = { can_start_scheme = { type = mm_wyvern_tame_fierce_captive target_character = scope:recipient } }
		}
	}

	on_accept = {
		scope:actor = {
			if = {
				limit = { scope:recipient = { has_trait = myv_wyvern_temperament_alert } }
				start_scheme = { type = mm_wyvern_tame_alert_captive target_character = scope:recipient }
			}
			else_if = {
				limit = { scope:recipient = { has_trait = myv_wyvern_temperament_docile } }
				start_scheme = { type = mm_wyvern_tame_docile_captive target_character = scope:recipient }
			}
			else_if = {
				limit = { scope:recipient = { has_trait = myv_wyvern_temperament_fierce } }
				start_scheme = { type = mm_wyvern_tame_fierce_captive target_character = scope:recipient }
			}
		}
	}
}
