mm_wyvern_is_legendary_blood = {
    OR = { has_trait = myv_wyvern_legendary has_trait = mm_wyvern_legendary_origin has_character_flag = mm_wyvern_legendary_origin }
}

mm_wyvern_is_very_hungry = {
    myv_is_living_wyvern = yes
    OR = {
        AND = { has_variable = myv_wyvern_hunger var:myv_wyvern_hunger < 10 }
        myv_wyvern_malnutrition_progress > 0
    }
}

# Assigned reciprocal mounts remain reserved for the existing rider system.
mm_wyvern_has_assigned_rider = {
    trigger_if = {
        limit = { exists = var:myv_war_mount_rider }
        var:myv_war_mount_rider = {
            is_alive = yes
            has_trait = myv_wyvern_rider
            trigger_if = {
                limit = { exists = var:myv_war_mount }
                var:myv_war_mount = prev
            }
            trigger_else = { always = no }
        }
    }
    trigger_else = { always = no }
}

mm_wyvern_can_serve_as_knight = {
    is_available_quick = { ai = yes adult = yes imprisoned = no incapable = no hostage = no }
    myv_is_living_wyvern = yes
    has_trait = myv_wyvern_tamed
    exists = employer
    employer = $ARMY_OWNER$
    NOT = { has_variable = mm_wyvern_captor }
    NOT = { mm_wyvern_has_assigned_rider = yes }
    NOT = { has_character_flag = myv_wyvern_travelling_with_rider }
    NOT = { exists = current_travel_plan }
    trigger_if = {
        limit = { exists = var:myv_wyvern_bonded_character }
        var:myv_wyvern_bonded_character = { is_alive = yes }
    }
    trigger_else = { always = no }
    trigger_if = {
        limit = { $ARMY_OWNER$ = { is_ai = no } }
        is_knight_permitted = $ARMY_OWNER$
    }
}

mm_wyvern_is_independent_knight_deployed = {
    trigger_if = {
        limit = { exists = employer }
        mm_wyvern_can_serve_as_knight = { ARMY_OWNER = employer }
        exists = knight_army.location
    }
    trigger_else = { always = no }
}

# These thresholds are for the shared on-foot human finishing roll, not aerial duels.
mm_wyvern_hunter_can_finish = {
    myv_is_wyvern = no
    exists = scope:myv_slay_wyvern_target
    trigger_if = {
        limit = { scope:myv_slay_wyvern_target = { mm_wyvern_is_legendary_blood = yes } }
        prowess > 60
    }
    trigger_else = { prowess > 30 }
}

myv_is_wyvern_deployed_with_army = {
    OR = {
        mm_wyvern_is_independent_knight_deployed = yes
        AND = {
    trigger_if = {
        limit = { myv_is_living_wyvern = yes exists = var:myv_war_mount_rider }
        var:myv_war_mount_rider = {
            myv_is_active_wyvern_war_rider = yes
            var:myv_war_mount = prev
            OR = {
                exists = commanding_army.location
                exists = knight_army.location
            }
        }
    }
    trigger_else = { always = no }
        }
    }
}

can_be_knight_trigger = {
    trigger_if = {
        limit = { myv_is_wyvern = yes }
        mm_wyvern_can_serve_as_knight = { ARMY_OWNER = $ARMY_OWNER$ }
    }
    trigger_else = {
	is_available_quick = {
		ai = yes
		hostage = no
	}
	myv_is_wyvern = no
	can_be_warrior_trigger = { ARMY_OWNER = $ARMY_OWNER$ }
	trigger_if = { # Admin governors cannot be knights
		limit = {
			$ARMY_OWNER$ = { government_has_mechanic = administrative }
			is_courtier = no
			exists = primary_title
		}
		primary_title = {
			tier < tier_duchy
			trigger_if = {
				limit = { is_noble_family_title = yes }
				always = no
			}
		}
	}
	trigger_if = {
		limit = {
			$ARMY_OWNER$.culture = {
				has_cultural_parameter = minimum_prowess_for_knights
			}
		}
		prowess >= 12
	}
	trigger_if = {
		limit = {
			is_clergy = yes
		}
		OR = {
			rite = { rite_has_parameter = clergy_can_fight }
			AND = {
				has_personal_tenet_flag = clergy_can_fight_personal
				NOT = { rite = { rite_has_parameter = pacifist } }
			}
			culture = { has_cultural_parameter = culture_clergy_can_fight }
			AND = {
				$ARMY_OWNER$.culture = {
					has_cultural_parameter = high_prowess_ignores_knight_restrictions
				}
				prowess >= 10
			}
		}
	}
	OR = {
		AND = {
			$ARMY_OWNER$.culture = {
				has_cultural_parameter = high_prowess_ignores_knight_restrictions
			}
			prowess >= 10
		}
		has_trait = order_member
		NOR = {
			has_trait = blind
			has_court_position = master_assassin_court_position
		}
	}
	trigger_if = {
		limit = {
			exists = liege.diarch
		}
		this != liege.diarch
	}
	trigger_if = {
		limit = {
			$ARMY_OWNER$ = {
				is_ai = no
			}
		}
		is_knight_permitted = $ARMY_OWNER$
	}
    }
}
