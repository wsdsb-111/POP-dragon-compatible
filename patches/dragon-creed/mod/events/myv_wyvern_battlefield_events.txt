# The player and the designated war mount share the left portrait scene; the
# spotted enemy commander occupies the right. The mounted portrait variable is
# removed after the event so ordinary character portraits remain untouched.
myv_wyvern_battlefield.0001 = {
	type = character_event
	window = myv_wyvern_dive_event
	override_background = { reference = myv_wyvern_dive }
	title = myv_wyvern_battlefield.0001.t
	desc = myv_wyvern_battlefield.0001.desc
	theme = martial

	left_portrait = {
		character = root
		animation = myv_wyvern_rider_flying
		outfit_tags = { military_outfit }
	}
	right_portrait = {
		character = root.var:myv_battlefield_event_enemy_commander
		animation = shock
		outfit_tags = { military_outfit }
	}

	immediate = {
		# Keep the tooltip target independent of variables cleared by option previews.
		var:myv_battlefield_event_enemy_commander = { save_scope_as = myv_battlefield_tooltip_enemy }
		set_variable = {
			name = myv_riding_portrait_wyvern
			value = var:myv_battlefield_event_wyvern
		}
		# The left portrait character is human, so mirror the mount's size tier
		# onto temporary rider flags for the size-aware mirrored dive camera.
		var:myv_battlefield_event_wyvern = {
			if = { limit = { has_character_flag = myv_wyvern_size_hatchling } root = { add_character_flag = myv_battlefield_mount_size_hatchling } }
			else_if = { limit = { has_character_flag = myv_wyvern_size_tier_01 } root = { add_character_flag = myv_battlefield_mount_size_tier_01 } }
			else_if = { limit = { has_character_flag = myv_wyvern_size_tier_02 } root = { add_character_flag = myv_battlefield_mount_size_tier_02 } }
			else_if = { limit = { has_character_flag = myv_wyvern_size_tier_03 } root = { add_character_flag = myv_battlefield_mount_size_tier_03 } }
			else_if = { limit = { has_character_flag = myv_wyvern_size_tier_04 } root = { add_character_flag = myv_battlefield_mount_size_tier_04 } }
			else_if = { limit = { has_character_flag = myv_wyvern_size_tier_05 } root = { add_character_flag = myv_battlefield_mount_size_tier_05 } }
			else_if = { limit = { has_character_flag = myv_wyvern_size_tier_06 } root = { add_character_flag = myv_battlefield_mount_size_tier_06 } }
			else_if = { limit = { has_character_flag = myv_wyvern_size_tier_07 } root = { add_character_flag = myv_battlefield_mount_size_tier_07 } }
			else_if = { limit = { has_character_flag = myv_wyvern_size_tier_08 } root = { add_character_flag = myv_battlefield_mount_size_tier_08 } }
			else_if = { limit = { has_character_flag = myv_wyvern_size_tier_09 } root = { add_character_flag = myv_battlefield_mount_size_tier_09 } }
			else_if = { limit = { has_character_flag = myv_wyvern_size_tier_10 } root = { add_character_flag = myv_battlefield_mount_size_tier_10 } }
		}
	}

	option = {
		name = myv_wyvern_battlefield.0001.a
		trigger = {
			var:myv_battlefield_event_enemy_commander = {
				is_alive = yes
				NOT = { myv_is_active_wyvern_war_rider = yes }
			}
		}
        custom_tooltip = mm_creed_bf_pounce_tt
        hidden_effect = {
            var:myv_battlefield_event_enemy_commander = {
                mm_creed_earth_wyvern_return_roll_effect = yes
                if = { limit = { has_character_flag = mm_creed_earth_wyvern_returned } remove_character_flag = mm_creed_earth_wyvern_returned }
                else = { myv_record_enemy_rider_battle_death_effect = yes death = { death_reason = death_myv_wyvern_battlefield_pounce killer = root } root = { add_prestige = knight_prestige_gain_on_kill } }
            }
        }
	}

	option = {
		name = myv_wyvern_battlefield.0001.c
		trigger = {
			var:myv_battlefield_event_enemy_commander = {
				myv_is_active_wyvern_war_rider = yes
			}
		}
		hidden_effect = {
			set_variable = {
				name = myv_wyvern_duel_enemy_rider
				value = var:myv_battlefield_event_enemy_commander
			}
			myv_begin_wyvern_rider_duel_effect = yes
		}
	}

	option = {
		name = myv_wyvern_battlefield.0001.b
	}

	after = {
		remove_variable = myv_battlefield_event_wyvern
		# The mounted-duel result still needs these battle scopes to attribute
		# deaths correctly, and it also reuses the portrait-mount variable. The
		# duel cleanup removes all of them after its third round.
		if = {
			limit = { NOT = { has_variable = myv_wyvern_duel_enemy_rider } }
			remove_variable = myv_riding_portrait_wyvern
			remove_variable = myv_battlefield_event_enemy_commander
			remove_variable = myv_battlefield_event_enemy_war_leader
			remove_variable = myv_battlefield_event_friendly_war_leader
			remove_variable = myv_battlefield_event_location
		}
		remove_character_flag = myv_wyvern_battlefield_event_portrait
		remove_character_flag = myv_battlefield_mount_size_hatchling
		remove_character_flag = myv_battlefield_mount_size_tier_01
		remove_character_flag = myv_battlefield_mount_size_tier_02
		remove_character_flag = myv_battlefield_mount_size_tier_03
		remove_character_flag = myv_battlefield_mount_size_tier_04
		remove_character_flag = myv_battlefield_mount_size_tier_05
		remove_character_flag = myv_battlefield_mount_size_tier_06
		remove_character_flag = myv_battlefield_mount_size_tier_07
		remove_character_flag = myv_battlefield_mount_size_tier_08
		remove_character_flag = myv_battlefield_mount_size_tier_09
		remove_character_flag = myv_battlefield_mount_size_tier_10
	}
}
