mm_v1312_silk_road_queue_daily_effect = {
    save_scope_as = mm_v1312_silk_road_target
    if = {
        limit = {
            mm_v1312_silk_sea_member_trigger = yes
            mm_v1312_silk_road_active_participant_trigger = yes
            NOT = { has_character_flag = mm_v1312_silk_road_daily_queued }
        }
        add_character_flag = { flag = mm_v1312_silk_road_daily_queued days = 2 }
        trigger_event = { id = mm_v1312_silk_road.0001 days = 1 }
    }
}

mm_v1312_silk_road_region_bonus_effect = {
    if = {
        limit = { mm_v1312_silk_road_region_bounty_high_tax_trigger = { SUB_REGION = $SUB_REGION$ } }
        add_character_modifier = { modifier = $BOUNTY_MODIFIER$ days = 2 }
    }
    else_if = {
        limit = { mm_v1312_silk_road_region_bounty_periphery_tax_trigger = { SUB_REGION = $SUB_REGION$ } }
        add_character_modifier = { modifier = $PERIPHERY_MODIFIER$ days = 2 }
    }
    else_if = {
        limit = { mm_v1312_silk_road_region_steady_tax_trigger = { SUB_REGION = $SUB_REGION$ } }
        add_character_modifier = { modifier = $STEADY_MODIFIER$ days = 2 }
    }
}
