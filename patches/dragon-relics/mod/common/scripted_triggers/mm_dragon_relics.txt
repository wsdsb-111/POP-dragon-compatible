mm_relic_black_crown_equipped_trigger = {
    OR = { AND = { exists = faith:mm_dragon_creed exists = rite:mm_dragon_black trigger_if = { limit = { exists = faith:mm_dragon_creed exists = rite:mm_dragon_black } faith = faith:mm_dragon_creed rite = rite:mm_dragon_black } } has_character_modifier = mm_relic_fire_forging }
    any_equipped_character_artifact = { has_variable = mm_relic_black_crown_identity }
}
mm_relic_black_crown_ready_trigger = {
    is_alive = yes
    faith = faith:mm_dragon_creed
    rite = rite:mm_dragon_black
    mm_relic_black_crown_equipped_trigger = yes
    mm_relic_has_black_blood_trigger = yes
    any_equipped_character_artifact = {
        has_variable = mm_relic_black_crown_identity
        NOT = { has_variable = mm_relic_cooldown }
    }
}
