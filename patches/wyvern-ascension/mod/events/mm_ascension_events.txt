    type = character_event
    title = mm_asc.0001.t
    desc = mm_asc.0001.desc
    theme = mental_break
    trigger = { mm_asc_is_candidate = yes }
    option = {
        name = mm_asc.0001.a
        if = { limit = { mm_asc_is_candidate = yes } add_stress = 25 }
    }
    option = {
        name = mm_asc.0001.b
        if = {
            limit = { mm_asc_is_candidate = yes }

            else = { add_character_modifier = { modifier = mm_asc_whisper_paranoia years = 2 } }
        }
    }
}
