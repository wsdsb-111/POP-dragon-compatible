# Repair old completed bodies using their recorded blood source only.
mm_royal_repair_source_body_effect = {
    if = {
        limit = { has_trait = mm_asc_dragon_body exists = var:mm_asc_blood_source }
        if = {
            limit = { OR = { NOT = { has_variable = mm_asc_source_reference_size } var:mm_asc_source_reference_size <= 0 } var:mm_asc_blood_source = { mm_royal_reference_size > 0 } }
            set_variable = { name = mm_asc_source_reference_size value = var:mm_asc_blood_source.mm_royal_reference_size }
        }
        if = {
            limit = { OR = { NOT = { has_variable = mm_asc_source_reference_mass } var:mm_asc_source_reference_mass <= 0 } var:mm_asc_blood_source = { mm_royal_reference_mass > 0 } }
            set_variable = { name = mm_asc_source_reference_mass value = var:mm_asc_blood_source.mm_royal_reference_mass }
        }
        if = {
            limit = { OR = { NOT = { has_variable = mm_asc_source_visual_tier } var:mm_asc_source_visual_tier < 0 var:mm_asc_source_visual_tier > 10 } }
            set_variable = { name = mm_asc_source_visual_tier value = var:mm_asc_blood_source.myv_wyvern_visual_size_tier }
        }
        if = {
            limit = { OR = { NOT = { has_variable = mm_royal_visual_tier } var:mm_royal_visual_tier < 0 var:mm_royal_visual_tier > 10 } }
            set_variable = { name = mm_royal_visual_tier value = var:mm_asc_source_visual_tier }
            myv_refresh_wyvern_portrait_flags_effect = yes
        }
    }
}
