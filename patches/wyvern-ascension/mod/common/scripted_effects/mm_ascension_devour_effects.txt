# Survivor scope. All entry points revalidate the same current living pair.
mm_asc_unite_twins_effect = {
    if = {
        limit = { mm_asc_pair_can_unite = yes }
        save_scope_as = mm_royal_survivor
        var:mm_royal_twin = { save_scope_as = mm_royal_victim }
                    if = {
                        limit = { var:mm_royal_throne = 1 NOT = { has_global_variable = mm_royal_fire_united } }
                        set_global_variable = { name = mm_royal_fire_united value = yes }
                        set_variable = { name = mm_royal_consumed_twin value = scope:mm_royal_victim }

                        add_trait = mm_royal_complete
                        change_first_name = mm_royal_complete_name_fire
                        give_nickname = nick_mm_royal_complete_fire
                        scope:mm_royal_victim = { death = { death_reason = death_mm_royal_devoured killer = scope:mm_royal_survivor } }
                        mm_royal_repair_profile_effect = yes
                        mm_royal_apply_physical_profile_effect = yes
                        myv_refresh_wyvern_prowess_effect = yes
                    }
                    if = {
                        limit = { var:mm_royal_throne = 2 NOT = { has_global_variable = mm_royal_earth_united } }
                        set_global_variable = { name = mm_royal_earth_united value = yes }
                        set_variable = { name = mm_royal_consumed_twin value = scope:mm_royal_victim }

                        change_first_name = mm_royal_complete_name_earth

                        limit = { var:mm_royal_throne = 3 NOT = { has_global_variable = mm_royal_sea_united } }
