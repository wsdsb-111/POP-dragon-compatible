on_game_start_after_lobby = { on_actions = { mm_asc_load } }
yearly_global_pulse = { on_actions = { mm_asc_yearly_pulse } }

mm_asc_load = {
    effect = {
        mm_asc_initialize_seats_effect = yes
        if = {
            limit = { has_global_variable_list = mm_asc_sovereigns }
            every_in_global_list = {
                variable = mm_asc_sovereigns
                limit = { is_alive = yes has_trait = mm_asc_dragon_body var:mm_asc_outcome = flag:dragon }
                mm_royal_queue_daily_effect = yes
            }
        }
    }
}

mm_asc_yearly_pulse = {
    effect = {
        if = {
            limit = { has_global_variable_list = mm_asc_candidates }
            every_in_global_list = {
                variable = mm_asc_candidates
                limit = { mm_asc_is_candidate = yes }
                mm_asc_yearly_effect = yes
            }
        }
    }
}
