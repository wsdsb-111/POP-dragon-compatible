# Ascension extends the public palette API without a kings -> ascension dependency.
mm_royal_visual_fire = {
  NOT = { has_character_flag = mm_wyvern_human_form }
  OR = { AND = { NOT = { portrait_has_trait_trigger = { TRAIT = mm_asc_dragon_body } } AND = { OR = { portrait_has_trait_trigger = { TRAIT = mm_royal_dragon_king } portrait_has_trait_trigger = { TRAIT = mm_royal_complete } } has_variable = mm_royal_throne var:mm_royal_throne = 1 } }
    AND = { portrait_has_trait_trigger = { TRAIT = mm_asc_dragon_body } has_character_flag = mm_asc_dragon_form has_variable = mm_asc_visual_family var:mm_asc_visual_family = 1 }
  }
}
