mm_royal_player_available = {
    is_alive = yes
    is_adult = yes
    is_imprisoned = no
    is_incapable = no
    OR = { myv_is_wyvern = no has_trait = mm_asc_dragon_body }
}

mm_asc_is_current_incarnation = {
    exists = var:mm_asc_seat_anchor
    var:mm_asc_seat_anchor = { exists = var:mm_asc_current_incarnation var:mm_asc_current_incarnation = prev }
}

# Survivor scope; used both by a keeper's command and a successor's own act.
mm_asc_pair_can_unite = {
    is_alive = yes
    has_trait = mm_royal_dragon_king
    mm_asc_is_current_incarnation = yes
    NOT = { has_trait = mm_royal_torpor }
    trigger_if = {
        limit = { exists = var:mm_royal_twin }
        NOT = { var:mm_royal_twin = this }
        has_relation_mm_dragon_twin = var:mm_royal_twin
        var:mm_royal_twin = {
            is_alive = yes
            has_trait = mm_royal_dragon_king
            mm_asc_is_current_incarnation = yes
            NOT = { has_trait = mm_royal_torpor }
            var:mm_royal_twin = prev
            var:mm_royal_throne = prev.var:mm_royal_throne
        }
    }
    trigger_else = { always = no }
    OR = {
        AND = { var:mm_royal_throne = 1 NOT = { has_global_variable = mm_royal_fire_united } }
        AND = { var:mm_royal_throne = 2 NOT = { has_global_variable = mm_royal_earth_united } }
        AND = { var:mm_royal_throne = 3 NOT = { has_global_variable = mm_royal_sea_united } }
        AND = { var:mm_royal_throne = 4 NOT = { has_global_variable = mm_royal_sky_united } }
    }
}

mm_royal_pair_ready_for_devour = {
    mm_royal_player_available = yes
    scope:recipient = {
        is_alive = yes
        has_trait = mm_royal_dragon_king
        mm_asc_is_current_incarnation = yes
        NOT = { has_trait = mm_royal_torpor }
        has_relation_myv_wyvern_bonded_character = scope:actor
        trigger_if = {
            limit = { exists = var:mm_royal_twin }
            has_relation_mm_dragon_twin = var:mm_royal_twin
            var:mm_royal_twin = {
                is_alive = yes
                has_trait = mm_royal_dragon_king
                mm_asc_is_current_incarnation = yes
                NOT = { has_trait = mm_royal_torpor }
                has_relation_myv_wyvern_bonded_character = scope:actor
                var:mm_royal_twin = prev
                var:mm_royal_throne = prev.var:mm_royal_throne
            }
        }
        trigger_else = { always = no }
    }
    OR = {
        AND = { scope:recipient = { var:mm_royal_throne = 1 } NOT = { has_global_variable = mm_royal_fire_united } }
        AND = { scope:recipient = { var:mm_royal_throne = 2 } NOT = { has_global_variable = mm_royal_earth_united } }
        AND = { scope:recipient = { var:mm_royal_throne = 3 } NOT = { has_global_variable = mm_royal_sea_united } }
        AND = { scope:recipient = { var:mm_royal_throne = 4 } NOT = { has_global_variable = mm_royal_sky_united } }
    }
}
